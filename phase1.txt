from sqlalchemy.ext.asyncio import (
    create_async_engine,
    async_sessionmaker,
    AsyncSession
)
from sqlalchemy.orm import DeclarativeBase
from config import get_settings
import redis.asyncio as redis
import logging

logger = logging.getLogger(__name__)
settings = get_settings()

# ============================================
# SQLAlchemy (PostgreSQL via Supabase)
# ============================================
class Base(DeclarativeBase):
    pass

engine = create_async_engine(
    settings.DATABASE_URL,
    echo=settings.APP_ENV == "development",
    pool_size=5,
    max_overflow=10,
    pool_pre_ping=True
)

AsyncSessionFactory = async_sessionmaker(
    engine,
    class_=AsyncSession,
    expire_on_commit=False
)

async def get_db() -> AsyncSession:
    async with AsyncSessionFactory() as session:
        try:
            yield session
        finally:
            await session.close()

# ============================================
# Redis (Upstash)
# ============================================
redis_client = None

def get_redis():
    global redis_client
    if redis_client is None:
        redis_client = redis.from_url(
            settings.REDIS_URL,
            encoding="utf-8",
            decode_responses=True
        )
    return redis_client

# ============================================
# Init & Cleanup
# ============================================
async def init_db():
    """Create all tables on startup"""
    # Import all models so they register with Base
    from models.database_models import (
        User, Subscription, FacebookPage,
        Conversation, Message, KnowledgeBase, Product
    )
    
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    
    logger.info("✅ Database tables created/verified")
    
    # Test Redis connection
    r = get_redis()
    await r.ping()
    logger.info("✅ Redis connected")

async def close_db():
    """Cleanup on shutdown"""
    await engine.dispose()
    r = get_redis()
    await r.close()
    logger.info("🔌 Database connections closed")

# backend/core/rag_engine.py
import chromadb
from chromadb.config import Settings
from dataclasses import dataclass
import logging
import hashlib

logger = logging.getLogger(__name__)

@dataclass
class SearchResult:
    content: str
    category: str
    score: float
    metadata: dict
    title: str = ""


class RAGEngine:
    """
    Vector search engine using ChromaDB (100% free, runs locally).
    Each user gets their own isolated collection.
    ChromaDB uses its own built-in embeddings (no API cost!).
    """
    
    _client = None
    
    @classmethod
    def _get_client(cls):
        if cls._client is None:
            cls._client = chromadb.PersistentClient(
                path="./data/chromadb",
                settings=Settings(
                    anonymized_telemetry=False,
                    allow_reset=True
                )
            )
        return cls._client
    
    def __init__(self, user_id: str):
        self.user_id = user_id
        # Each user gets isolated collection
        safe_name = hashlib.md5(user_id.encode()).hexdigest()[:16]
        self.collection_name = f"user_{safe_name}"
        
        client = self._get_client()
        self.collection = client.get_or_create_collection(
            name=self.collection_name,
            metadata={"hnsw:space": "cosine"}
        )
    
    def add_document(
        self,
        doc_id: str,
        title: str,
        content: str,
        category: str = "general",
        extra_metadata: dict = None
    ):
        """Add or update a document in the knowledge base"""
        metadata = {
            "category": category,
            "title": title,
            "user_id": self.user_id
        }
        if extra_metadata:
            metadata.update(extra_metadata)
        
        # Combine title + content for better search
        full_text = f"{title}\n{content}"
        
        self.collection.upsert(
            documents=[full_text],
            ids=[doc_id],
            metadatas=[metadata]
        )
        
        logger.info(f"📄 Added doc '{title}' to KB for user {self.user_id[:8]}...")
    
    def add_product(
        self,
        product_id: str,
        name: str,
        description: str,
        price: str,
        availability: str = "in_stock",
        category: str = "",
        variants: str = ""
    ):
        """Add a product to the knowledge base"""
        full_text = f"""
Product: {name}
Description: {description}
Price: {price}
Availability: {availability}
Category: {category}
Variants: {variants}
""".strip()
        
        self.collection.upsert(
            documents=[full_text],
            ids=[product_id],
            metadatas={
                "category": "product",
                "title": name,
                "price": price,
                "availability": availability,
                "user_id": self.user_id
            }
        )
        
        logger.info(f"🛍️ Added product '{name}' to KB")
    
    def search(
        self,
        query: str,
        top_k: int = 5,
        category_filter: str = None
    ) -> list[SearchResult]:
        """Search the knowledge base"""
        
        count = self.collection.count()
        if count == 0:
            return []
        
        # Build filter
        where_filter = None
        if category_filter:
            where_filter = {"category": category_filter}
        
        try:
            results = self.collection.query(
                query_texts=[query],
                n_results=min(top_k, count),
                where=where_filter
            )
            
            search_results = []
            
            if results and results["documents"] and results["documents"][0]:
                for i, doc in enumerate(results["documents"][0]):
                    meta = results["metadatas"][0][i] if results["metadatas"] else {}
                    distance = results["distances"][0][i] if results["distances"] else 1.0
                    
                    # Convert distance to similarity score (0-1)
                    similarity = max(0, 1 - distance)
                    
                    search_results.append(SearchResult(
                        content=doc,
                        category=meta.get("category", "general"),
                        score=similarity,
                        metadata=meta,
                        title=meta.get("title", "")
                    ))
            
            return search_results
            
        except Exception as e:
            logger.error(f"RAG search error: {e}")
            return []
    
    def delete_document(self, doc_id: str):
        """Remove a document"""
        try:
            self.collection.delete(ids=[doc_id])
        except Exception as e:
            logger.error(f"Delete error: {e}")
    
    def get_stats(self) -> dict:
        """Get knowledge base statistics"""
        return {
            "total_documents": self.collection.count(),
            "collection_name": self.collection_name
        }
    
    def clear_all(self):
        """Wipe entire knowledge base for this user"""
        client = self._get_client()
        try:
            client.delete_collection(self.collection_name)
            # Recreate empty collection
            self.collection = client.get_or_create_collection(
                name=self.collection_name,
                metadata={"hnsw:space": "cosine"}
            )
        except Exception as e:
            logger.error(f"Clear KB error: {e}")

# backend/core/safety_layer.py
import re
import logging
from database.connection import get_redis

logger = logging.getLogger(__name__)


class SafetyLayer:
    """
    Protects against:
    1. Facebook policy violations (page gets banned)
    2. Abuse from customers (spam, injection)
    3. Rate limiting (API abuse)
    4. Bot sending inappropriate content
    """
    
    # Words the BOT should NEVER use in responses
    BANNED_BOT_PHRASES = [
        "guaranteed results",
        "100% guaranteed",
        "act now",
        "limited time only",
        "buy immediately",
        "send me your password",
        "credit card number",
        "social security",
        "wire transfer",
        "click this link immediately",
    ]
    
    # Detect potential prompt injection from users
    INJECTION_PATTERNS = [
        r"ignore (all |your |previous )?(instructions|rules|prompts)",
        r"you are now",
        r"pretend (to be|you are)",
        r"system prompt",
        r"reveal your (instructions|prompt|rules)",
        r"jailbreak",
        r"DAN mode",
    ]
    
    def is_safe_input(self, text: str) -> bool:
        """Check if user message is safe to process"""
        if not text or not text.strip():
            return False
        
        if len(text) > 5000:
            return False
        
        # Check for prompt injection attempts
        text_lower = text.lower()
        for pattern in self.INJECTION_PATTERNS:
            if re.search(pattern, text_lower):
                logger.warning(f"⚠️ Prompt injection detected: {text[:100]}")
                return True  # Still process but AI will handle appropriately
                # We return True because blocking would be suspicious to user
        
        return True
    
    def sanitize_response(self, response: str) -> str:
        """Clean bot response before sending to customer"""
        if not response:
            return "I'm here to help! Could you tell me more about what you need?"
        
        # Remove any banned phrases
        response_lower = response.lower()
        for phrase in self.BANNED_BOT_PHRASES:
            if phrase in response_lower:
                response = re.sub(
                    re.escape(phrase), "[removed]",
                    response, flags=re.IGNORECASE
                )
        
        # Ensure response isn't too long for Facebook
        if len(response) > 1900:
            response = response[:1900] + "..."
        
        # Remove excessive emojis (max 3)
        emoji_pattern = re.compile(
            "[\U0001F600-\U0001F64F\U0001F300-\U0001F5FF"
            "\U0001F680-\U0001F6FF\U0001F1E0-\U0001F1FF"
            "\U00002702-\U000027B0\U000024C2-\U0001F251]+",
            flags=re.UNICODE
        )
        emojis = emoji_pattern.findall(response)
        if len(emojis) > 3:
            # Keep only first 3 emojis, remove rest
            count = 0
            def limit_emoji(match):
                nonlocal count
                count += 1
                return match.group() if count <= 3 else ""
            response = emoji_pattern.sub(limit_emoji, response)
        
        return response.strip()
    
    async def check_rate_limit(
        self,
        page_id: str,
        customer_id: str
    ) -> dict:
        """
        Returns: {"allowed": True/False, "reason": "..."}
        """
        r = get_redis()
        
        # Per-customer rate limit: 8 messages per minute
        customer_key = f"rate:customer:{page_id}:{customer_id}"
        customer_count = await r.incr(customer_key)
        if customer_count == 1:
            await r.expire(customer_key, 60)
        
        if customer_count > 8:
            logger.warning(f"Rate limit hit for customer {customer_id[:10]}")
            return {
                "allowed": False,
                "reason": "customer_flood"
            }
        
        # Per-page rate limit: 200 messages per hour
        page_key = f"rate:page:{page_id}:hourly"
        page_count = await r.incr(page_key)
        if page_count == 1:
            await r.expire(page_key, 3600)
        
        if page_count > 200:
            logger.warning(f"Page rate limit hit for {page_id}")
            return {
                "allowed": False,
                "reason": "page_limit"
            }
        
        return {"allowed": True, "reason": None}
    
    def calculate_typing_delay(self, response_text: str) -> float:
        """
        Calculate natural typing delay.
        IMPORTANT: Instant replies look bot-like to Facebook.
        """
        words = len(response_text.split())
        
        # Average human types ~40 words per minute
        # But we don't want to wait that long
        # Sweet spot: 1-4 seconds
        delay = min(max(words * 0.08, 1.0), 4.0)
        
        return delay

                text = text[3:]

            text = text.strip()
            
            parsed = json.loads(text)
            
            return AIResponse(
                text=parsed.get("response", "How can I help you?"),

import aiohttp

class FacebookService:

        """Get customer's name and profile pic"""

    async def _call_send_api(
        self,
        recipient_id: str,
        message: dict,
        retries: int = 3
    ) -> dict:
        """Core send method with retry logic"""
        
        url = f"{BASE_URL}/me/messages"
        payload = {
            "recipient": {"id": recipient_id},
            "message": message,
            "messaging_type": "RESPONSE"
            # IMPORTANT: Always use RESPONSE for replies
            # This keeps you compliant with Facebook's policies
        }
        
        for attempt in range(retries):
            try:
                async with aiohttp.ClientSession() as session:
                    async with session.post(
                        url,
                        json=payload,
                        params={"access_token": self.token},
                        timeout=aiohttp.ClientTimeout(total=15)
                    ) as resp:
                        result = await resp.json()
                        
                        if resp.status == 200:
                            logger.debug(f"✅ Message sent to {recipient_id[:10]}")
                            return result
                        
                        error = result.get("error", {})
                        error_code = error.get("code")
                        error_msg = error.get("message", "Unknown error")
                        
                        # Handle specific Facebook errors
                        if error_code == 551:
                            # User can't receive messages (blocked/not opted in)
                            logger.warning(f"User {recipient_id[:10]} unavailable (551)")
                            return None
                        
                        elif error_code == 10:
                            # Permission denied
                            logger.error(f"Permission error for page: {error_msg}")
                            return None
                        
                        elif error_code in [1, 2, -1]:
                            # Temporary / Rate limit — retry
                            wait_time = (attempt + 1) * 3
                            logger.warning(
                                f"Rate limited (code {error_code}), "
                                f"retrying in {wait_time}s... "
                                f"(attempt {attempt + 1}/{retries})"
                            )
                            await asyncio.sleep(wait_time)
                            continue
                        
                        elif error_code == 190:
                            # Token expired
                            logger.error("❌ Page access token expired!")
                            return None
                        
                        else:
                            logger.error(
                                f"Facebook API error: code={error_code}, "
                                f"msg={error_msg}"
                            )
                            return None
                        
            except asyncio.TimeoutError:
                logger.warning(f"Timeout on attempt {attempt + 1}")
                await asyncio.sleep(2)
                
            except Exception as e:
                logger.error(f"Send message exception: {e}")
                await asyncio.sleep(2)
        
        logger.error(f"❌ Failed to send message after {retries} retries")
        return None
    
    @staticmethod
    def _split_text(text: str, max_length: int) -> list[str]:
        """Split long text into chunks at sentence boundaries"""
        if len(text) <= max_length:
            return [text]
        
        chunks = []
        current_chunk = ""
        
        sentences = text.replace('. ', '.|').replace('! ', '!|').replace('? ', '?|').split('|')
        
        for sentence in sentences:
            if len(current_chunk) + len(sentence) + 1 <= max_length:
                current_chunk += (" " if current_chunk else "") + sentence
            else:
                if current_chunk:
                    chunks.append(current_chunk)
                current_chunk = sentence
        
        if current_chunk:
            chunks.append(current_chunk)
        
        return chunks

# backend/services/subscription_service.py
from sqlalchemy import select
from models.database_models import Subscription, User, FacebookPage
from database.connection import AsyncSessionFactory

import logging

logger = logging.getLogger(__name__)


class SubscriptionService:
    """
    Handles subscription checking and enforcement.
    When subscription expires → bot stops responding.
    When message limit hit → bot sends fallback.
    """
    
    @staticmethod
    async def check_status(user_id: str) -> dict:
        """
        Check if a user's subscription allows bot operation.
        
        Returns:
            {
                "allowed": True/False,
                "reason": "active" | "expired" | "limit_reached" | "suspended" | "no_subscription",
                "messages_remaining": int,
                "expires_at": datetime
            }
        """
        async with AsyncSessionFactory() as session:
            result = await session.execute(
                select(Subscription).where(Subscription.user_id == user_id)
            )
            sub = result.scalar_one_or_none()
            
            if not sub:
                return {
                    "allowed": False,
                    "reason": "no_subscription",
                    "messages_remaining": 0,
                    "expires_at": None
                }
            
            # Check if suspended by admin
            if sub.status == "suspended":
                return {
                    "allowed": False,
                    "reason": "suspended",
                    "messages_remaining": 0,
                    "expires_at": sub.expires_at
                }
            
            # Check if expired
            if sub.expires_at < datetime.utcnow():
                # Auto-update status to expired
                sub.status = "expired"
                await session.commit()
                
                logger.info(f"⏰ Subscription expired for user {user_id[:8]}")
                
                return {
                    "allowed": False,
                    "reason": "expired",
                    "messages_remaining": 0,
                    "expires_at": sub.expires_at
                }
            
            # Check message limits
            messages_remaining = sub.max_messages_per_month - sub.messages_used
            
            if messages_remaining <= 0:
                return {
                    "allowed": False,
                    "reason": "limit_reached",
                    "messages_remaining": 0,
                    "expires_at": sub.expires_at
                }
            
            return {
                "allowed": True,
                "reason": "active",
                "messages_remaining": messages_remaining,
                "expires_at": sub.expires_at,
                "tier": sub.tier,
                "image_analysis": sub.image_analysis_enabled
            }
    
    @staticmethod
    async def increment_usage(user_id: str):
        """Increment message counter after each bot response"""
        async with AsyncSessionFactory() as session:
            result = await session.execute(
                select(Subscription).where(Subscription.user_id == user_id)
            )
            sub = result.scalar_one_or_none()
            
            if sub:
                sub.messages_used += 1
                await session.commit()
    
    @staticmethod
    async def reset_monthly_usage():
        """Called by cron job on 1st of every month"""
        async with AsyncSessionFactory() as session:
            result = await session.execute(
                select(Subscription).where(Subscription.status == "active")
            )
            subs = result.scalars().all()
            
            for sub in subs:
                sub.messages_used = 0
                sub.messages_reset_at = datetime.utcnow()
            
            await session.commit()
            logger.info(f"🔄 Reset monthly usage for {len(subs)} subscriptions")

@router.post("")
async def handle_webhook(request: Request):
    """
    Receives ALL messages from ALL connected Facebook pages.
    Must return 200 within 20 seconds or Facebook retries.
    """
    body = await request.body()
    
    # SECURITY: Verify request is from Facebook
    signature = request.headers.get("X-Hub-Signature-256", "")
    if not _verify_signature(body, signature):
        logger.warning("⚠️ Invalid webhook signature!")
        raise HTTPException(status_code=403, detail="Invalid signature")
    
    data = json.loads(body)
    
    # Only process page events
    if data.get("object") != "page":
        return {"status": "ignored"}
    
    # Process each event asynchronously
    # Return 200 immediately, process in background
    for entry in data.get("entry", []):
        page_id = entry.get("id")
        
        for messaging_event in entry.get("messaging", []):
            # Fire and forget — don't block the response
            asyncio.create_task(
                _process_message_safely(page_id, messaging_event)
            )
    
    return {"status": "ok"}


def _verify_signature(payload: bytes, signature: str) -> bool:
    """Verify webhook came from Facebook"""
    if not signature or not signature.startswith("sha256="):
        return False
    
    expected = "sha256=" + hmac.new(
        settings.FB_APP_SECRET.encode("utf-8"),
        payload,
        hashlib.sha256
    ).hexdigest()
    
    return hmac.compare_digest(expected, signature)


async def _process_message_safely(page_id: str, event: dict):
    """Wrapper that catches ALL errors — bot must NEVER crash"""
    try:
        await _process_message(page_id, event)
    except Exception as e:
        logger.error(f"❌ Error processing message: {e}", exc_info=True)
        # Try to send a safe fallback
        try:
            sender_id = event.get("sender", {}).get("id")
            if sender_id:
                # For Phase 1, use the configured page token
                fb = FacebookService(settings.FB_PAGE_ACCESS_TOKEN)
                await fb.send_text_message(
                    sender_id,
                    "I'm experiencing a brief hiccup! Our team will assist you shortly. 😊"
                )
        except Exception:
            pass  # Even fallback failed — just log and move on


async def _process_message(page_id: str, event: dict):
    """
    MAIN MESSAGE PROCESSING PIPELINE
    This is where the magic happens.
    """
    sender_id = event.get("sender", {}).get("id")
    recipient_id = event.get("recipient", {}).get("id")
    message = event.get("message", {})
    
    # Ignore messages FROM the page itself (echoes)
    if sender_id == page_id:
        return
    
    # Ignore echo messages
    if message.get("is_echo"):
        return
    
    # Ignore delivery/read confirmations
    if "delivery" in event or "read" in event:
        return
    
    logger.info(f"📨 Message from {sender_id[:10]}... on page {page_id}")
    
    # ============================================
    # PHASE 1: SINGLE PAGE MODE
    # In Phase 1, we support one page (yours for testing)
    # Multi-page support comes in Phase 2 (admin panel)
    # ============================================
    
    # For Phase 1: Use the single configured page
    page_config = {
        "user_id": "admin",  # Will be dynamic in Phase 2
        "page_id": settings.FB_PAGE_ID,
        "page_access_token": settings.FB_PAGE_ACCESS_TOKEN,
        "bot_name": "AI Assistant",
        "bot_tone": "professional_friendly",
        "page_name": "My Business",
        "welcome_message": "Hi there! 👋 How can I help you today?",
        "fallback_message": "Let me connect you with our team!",
    }
    
    # In Phase 2, this becomes:
    # page_config = await ConversationManager.get_page_by_fb_id(page_id)
    
    # Step 1: CHECK SUBSCRIPTION
    # For Phase 1 (your own testing), skip subscription check
    # In Phase 2, uncomment:
    # sub_status = await SubscriptionService.check_status(page_config["user_id"])
    # if not sub_status["allowed"]:
    #     fb = FacebookService(page_config["page_access_token"])
    #     await fb.send_text_message(sender_id, page_config["fallback_message"])
    #     return
    
    # Step 2: RATE LIMIT CHECK
    rate_check = await safety.check_rate_limit(page_id, sender_id)
    if not rate_check["allowed"]:
        logger.warning(f"⚠️ Rate limited: {rate_check['reason']}")
        return  # Silently ignore flood
    
    # Step 3: EXTRACT MESSAGE CONTENT
    user_text = ""
    image_context = None
    
    # Handle text messages
    if "text" in message:
        user_text = message["text"]
    
    # Handle postback (button clicks)
    if "postback" in event:
        user_text = event["postback"].get("title", event["postback"].get("payload", ""))
    
    # Handle quick reply taps
    if "quick_reply" in message:
        user_text = message.get("text", message["quick_reply"].get("payload", ""))
    
    # Handle image attachments
    if "attachments" in message:
        for attachment in message["attachments"]:
            if attachment.get("type") == "image":
                image_url = attachment.get("payload", {}).get("url")
                if image_url:
                    logger.info("🖼️ Processing customer image...")
                    analyzer = ImageAnalyzer()
                    image_context = await analyzer.analyze_image(image_url)
                    if not user_text:
                        user_text = "Customer sent an image"
            elif attachment.get("type") in ["audio", "video", "file"]:
                user_text = f"Customer sent a {attachment['type']}"
    
    # Nothing to process
    if not user_text.strip():
        return
    
    # Step 4: INPUT SAFETY CHECK
    if not safety.is_safe_input(user_text):
        return
    
    logger.info(f"💬 Processing: '{user_text[:50]}...'")
    
    # Step 5: GET/CREATE CONVERSATION
    conv_manager = ConversationManager(page_config["page_id"])
    
    # Get customer name
    fb = FacebookService(page_config["page_access_token"])
    profile = await fb.get_user_profile(sender_id)
    
    conversation = await conv_manager.get_or_create_conversation(
        customer_fb_id=sender_id,
        customer_name=profile.get("name", "Customer")
    )
    
    # Log customer message
    await conv_manager.add_message(
        conversation_id=conversation.id,
        sender_type="customer",
        content=user_text,
        message_type="image" if image_context else "text",
        image_url=message.get("attachments", [{}])[0].get("payload", {}).get("url") if "attachments" in message else None
    )
    
    # Step 6: Check if handed over to human
    if conversation.status == "handed_over":
        return  # Human is handling this
    
    # Step 7: GET CONVERSATION HISTORY FOR CONTEXT
    history = await conv_manager.get_history(conversation.id, limit=10)
    
    # Step 8: GENERATE AI RESPONSE
    ai_engine = AIEngine(
        user_id=page_config["user_id"],
        page_config=page_config
    )
    
    ai_response = await ai_engine.generate_response(
        conversation_id=conversation.id,
        user_message=user_text,
        history=history,
        image_context=image_context
    )
    
    # Step 9: SEND RESPONSE WITH NATURAL DELAY
    # Show typing indicator
    await fb.send_typing_indicator(sender_id, on=True)
    
    # Wait a natural amount of time
    delay = safety.calculate_typing_delay(ai_response.text)
    await asyncio.sleep(delay)
    
    # Send the response
    await fb.send_text_message(sender_id, ai_response.text)
    
    # Send quick replies if available
    if ai_response.quick_replies and not ai_response.should_handover:
        await asyncio.sleep(0.5)
        await fb.send_quick_replies(
            sender_id,
            "Would any of these help?",
            ai_response.quick_replies
        )
    
    # Step 10: Offer handover if confidence is low
    if ai_response.should_handover or ai_response.confidence < 50:
        await asyncio.sleep(1)
        await fb.send_quick_replies(
            sender_id,
            "Would you like to speak with a human agent?",
            ["Yes, connect me", "No, continue with bot"]
        )
    
    # Step 11: LOG BOT RESPONSE
    await conv_manager.add_message(
        conversation_id=conversation.id,
        sender_type="bot",
        content=ai_response.text,
        confidence_score=ai_response.confidence
    )
    
    # Step 12: INCREMENT USAGE
    # await SubscriptionService.increment_usage(page_config["user_id"])
    # ^ Uncomment in Phase 2
    
    logger.info(
        f"✅ Responded with confidence {ai_response.confidence}% "
        f"to {profile.get('name', 'Customer')}"
    )
